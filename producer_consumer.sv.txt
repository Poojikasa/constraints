//1.You have two processes:Producer generates random 8-bit data.consumer reads data and checks if it is even or odd.Task (a) Write systemverilog code using a mailbox to implement producer-consumer communication.(b)Ensure the producer sends 10 random values and the consumer prints each value with its priority.
module producer_consumer_tb();
  mailbox #(bit[7:0])mbox;
  task producer();
    bit [7:0]data;
    repeat(10)
    begin
    data=$urandom_range(0,255);
    mbox.put(data);
      $display("[%0t]producer is sending data is %0d",$time,data);
              end
              endtask
              
              task consumer();
                bit [7:0]data;
                repeat(10)
                  begin
                    mbox.get(data);
                    if(data%2==0)
                      $display("[%0t]consumer received data is even", $time,data);
                    else
                      $display("[%0t]consumer received data is odd", $time,data);
                  end
              endtask
  initial
    begin
      mbox=new();
      fork
      producer();
      consumer();
      join
      $finish;
    end
endmodule

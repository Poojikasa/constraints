class dicegame;
rand int dice[5][5];
rand int total[5];
int max_score;
constraint c_dice{foreach 
(dice[i,j])
dice [i][j] inside {[1:6]};}
function void calculate();
max_score=0;
foreach(dice[i])
begin
total[i]=0;
foreach(dice[i][j])
begin
total[i]+=dice[i][j];
end
if(total[i]>max_score)
max_score=total[i];
end
endfunction
endclass
//endmodule
